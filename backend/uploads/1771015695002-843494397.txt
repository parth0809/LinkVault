

import math
import random
import re
import time
from collections import Counter, defaultdict

import numpy as np
import pandas as pd
import nltk
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer

TRAIN_PATH = "wiki_train.csv"
TEST_PATHS = {
    "Wiki": "wiki_test.csv",
    "Arxiv": "arxiv_test.csv",
    "Financial": "financial_test.csv",
}
VAL_SIZE = 100
RAND_SEED = 42
MIN_DOC_FREQ = 0.01  

nltk.download("stopwords", quiet=True)
nltk.download("wordnet", quiet=True)

STOP_WORDS = set(stopwords.words("english"))
wordnet = WordNetLemmatizer()

def get_text_column(df):
    cols = ["text", "abstract", "content"]
    lowered = {c.lower(): c for c in df.columns}
    for col in cols:
        if col in lowered:
            return lowered[col]

def generate_ngrams(tokens, n):
    if len(tokens) < n:
            return []
    return [tuple(tokens[i:i+n]) for i in range(len(tokens)-n+1)]

def load_documents(path):
    df = pd.read_csv(path)
    col = get_text_column(df)
    return df[col].dropna().astype(str).tolist()

def split_train_validation(documents):
    rng = random.Random(RAND_SEED)
    idx = list(range(len(documents)))
    rng.shuffle(idx)
    val_idx = set(idx[:VAL_SIZE])
    train = [documents[i] for i in range(len(documents)) if i not in val_idx]
    val = [documents[i] for i in range(len(documents)) if i in val_idx]
    return train, val

def preprocess_text(text):
    text = text.encode("ascii", errors="ignore").decode()
    text = re.sub(r"[^a-zA-Z\s]", " ", text)
    text = text.lower()
    tokens = text.split()
    return [wordnet.lemmatize(t) for t in tokens if t not in STOP_WORDS]


def add_start_and_end(tokens, n):
    return ["<s>"] * (n - 1) + tokens + ["</s>"]


def generate_ngrams(tokens, n):
    if len(tokens) < n:
        return []
    return [tuple(tokens[i:i+n]) for i in range(len(tokens)-n+1)]


def load_and_split_training_data(file_path, text_column):
    df = pd.read_csv(file_path)
    df = df.sample(frac=1, random_state=RAND_SEED).reset_index(drop=True)
    val = df.iloc[:VAL_SIZE]
    train = df.iloc[VAL_SIZE:]
    return train, val

def build_ngram_model(train, n):
    total_docs = len(train)
    min_docs = max(1, math.ceil(MIN_DOC_FREQ * total_docs))

    docs_ngrams = []
    doc_freq = Counter()

    for tokens in train:
        seq = add_start_and_end(tokens, n)
        grams = generate_ngrams(seq, n)
        docs_ngrams.append(grams)
        doc_freq.update(set(grams))

    vocab = {g for g, df in doc_freq.items() if df >= min_docs}

    counts = Counter()
    history_counts = Counter()
    context_ngram_counts = defaultdict(Counter)

    for grams in docs_ngrams:
        valid = [g for g in grams if g in vocab]
        counts.update(valid)
        if n == 1:
            history_counts[()] += len(valid)
        else:
            history_counts.update(g[:-1] for g in valid)
            for g in valid:
                context_ngram_counts[g[:-1]][g] += 1

    next_vocab = {g[-1] for g in vocab}
    cnt1 = Counter(counts.values())
    items = {h: Counter(c.values()) for h, c in context_ngram_counts.items()}

    return {
        "n": n,
        "vocab": vocab,
        "counts": counts,
        "history_counts": history_counts,
        "total_ngrams": sum(counts.values()),
        "next_vocab": max(1, len(next_vocab)),
        "counter": cnt1,
        "context": items,
    }



def laplace_prob(model, gram):
    n = model["n"]
    c = model["counts"].get(gram, 0)
    V = model["next_vocab"]

    if n == 1:
        denom = model["total_ngrams"] + V
    else:
        h = gram[:-1]
        denom = model["history_counts"].get(h, 0) + V
    return (c + 1) / denom if denom > 0 else 1.0 / V



def good_turing_prob(model, gram):
    n = model["n"]
    freq_table = model["counts"]
    count_value = freq_table.get(gram, 0)

    total_ngrams = model["total_ngrams"]
    if total_ngrams == 0:
        return 1e-12

    if n == 1:
        count_of_counts = model["counter"]
        N1 = count_of_counts.get(1, 0)
        if count_value == 0:
            return max((N1 / total_ngrams) if N1 > 0 else (1.0 / total_ngrams), 1e-12)
        Nc, Ncp1 = count_of_counts.get(count_value, 0), count_of_counts.get(count_value + 1, 0)
        tmp = (count_value + 1) * (Ncp1 / Nc) if Nc > 0 and Ncp1 > 0 else count_value
        return max(tmp / total_ngrams, 1e-12)

    history = gram[:-1]
    history_total = model["history_counts"].get(h, 0)
    global_N1 = model["counter"].get(1, 0)

    if history_total == 0:
        return max((global_N1 / total_ngrams) if global_N1 > 0 else (1.0 / total_ngrams), 1e-12)

    history_cont_of_cnt = model["context"].get(history, Counter())
    if count_value == 0:
        N1_history = history_cont_of_cnt.get(1, 0)
        if N1_history > 0:
            return max(N1_history / history_total, 1e-12)
        return max((global_N1 / total_ngrams) if global_N1 > 0 else (1.0 / total_ngrams), 1e-12)

    Nc, Ncp1 = history_cont_of_cnt.get(count_value, 0), history_cont_of_cnt.get(count_value + 1, 0)
    tmp = (count_value + 1) * (Ncp1 / Nc) if Nc > 0 and Ncp1 > 0 else count_value
    return max(tmp / history_total, 1e-12)


def article_perplexity(model, tokens, smoothing="laplace"):
    sequences = add_start_and_end(tokens, model["n"])
    grams = generate_ngrams(sequences, model["n"])
    if not grams:
        return float("inf")

    log_sum = 0.0
    for g in grams:
        p = laplace_prob(model, g) if smoothing == "laplace" else good_turing_prob(model, g)
        log_sum += math.log(max(p, 1e-12))

    return math.exp(-log_sum / len(grams))

def evaluate_dataset(model, docs_tokens, smoothing="laplace"):
    per_article = [article_perplexity(model, t, smoothing) for t in docs_tokens if len(t) > 0]
    return float(np.mean(per_article)), per_article

start_total = time.time()

train = load_documents(TRAIN_PATH)
train, val = split_train_validation(train)

t0 = time.time()
train_tokens = [preprocess_text(d) for d in train]
val_tokens = [preprocess_text(d) for d in val]
test_tokens = {
    name: [preprocess_text(d) for d in load_documents(path)]
    for name, path in TEST_PATHS.items()
}
print(f"Preprocessing time: {time.time() - t0:.2f}s")

results = defaultdict(dict)

#build and test across various time set
for n in [1, 2, 3]:
    t0 = time.time()
    model = build_ngram_model(train_tokens, n=n, min_doc_freq=MIN_DOC_FREQ)
    print(f"\nBuilt {n}-gram model in {(time.time()-t0):.2f}s | vocab={len(model['vocab'])}")

    for smoothing in ["laplace", "good_turing"]:
        val_avg, _ = evaluate_dataset(model, val_tokens, smoothing=smoothing)
        results["Validation"][f"{n}-gram {smoothing}"] = val_avg
        print(f"Validation | {n}-gram | {smoothing}: {val_avg:.4f}")

        for domain, docs in test_tokens.items():
            avg_pp, _ = evaluate_dataset(model, docs, smoothing=smoothing)
            results[domain][f"{n}-gram {smoothing}"] = avg_pp
            print(f"{domain:10s} | {n}-gram | {smoothing}: {avg_pp:.4f}")

print(f"\nTotal runtime: {time.time() - start_total:.2f}s")
